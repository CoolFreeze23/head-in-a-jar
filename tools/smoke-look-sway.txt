# The badge's swing on a turn (the look part): a hired cow turned by setup, every frame.
@world flat
@difficulty easy
@clean
time set 6000
gamerule doDaylightCycle false
gamerule doMobSpawning false
tp @s 0 -60 0 0 0
summon cow 0 -60 3 {Rotation:[180f,0f]}
give @s headinajar:staff_lanyard 2
wait 25
tp @s 0 -60 1.5 0 0
lookat cow 0.8
key use
wait 30
tp @s 2.5 -59.5 1 50 15
hud off
frames turn_cow 50
execute as @e[type=cow] at @s run tp @s ~ ~ ~ ~90 ~
wait 20
execute as @e[type=cow] at @s run tp @s ~ ~ ~ ~-90 ~
wait 30
hud on
