# Telekinesis films, shorter, for the shader and the owner-settings runs: a cow grabbed, swung round and thrown, from the jar's own eyes, then from a
# bystander at 5 and 30 blocks (the stand-in is the jar), by day and at night; then a grab at nothing.
@observe 20
wait 40
/time set 6000
/gamerule doMobSpawning false
/gamerule doDaylightCycle false
/weather clear
hud off
# --- the jar's own eyes: the real player is the jar
/tp @s 0.5 -60 0.5 0 0
jar
# the body stays where the head came off: the jar moves 3 blocks east of it
/tp @s 3.5 -60 0.5 0 0
fit telekinesis
charge 100000
wait 10
/summon cow 3.5 -60 4.5 {attributes:[{id:"minecraft:generic.movement_speed",base:0d}]}
wait 20
lookat cow
view 0
perf idle_jar_no_tether 40
wait 45
frames tk_doer_day 150
lookat cow
wait 10
key ability
wait 10
fxstate doer_grab
wait 30
perf doer_holding 40
turn 70 0 20
wait 12
turn -120 -5 30
wait 10
key attack
wait 50
fxstate after_doer_throw
# --- back to a body; the stand-in is the jar
body
wait 10
/kill @e[type=cow]
/kill @e[type=item]
bystander spawn 10 -60 0 0
bystander tick on
wait 10
bystander form jar
# the head comes off where the body stands: move the jar out of its own body
bystander tp 12 -60 0 0 0
bystander fit telekinesis
bystander charge 100000
/summon cow 12.5 -60 4.5 {attributes:[{id:"minecraft:generic.movement_speed",base:0d}]}
wait 30
bystander look 0 0
# 5 blocks, three-quarter, by day
/tp @s 16.5 -60 6.5 135 -2
wait 10

wait 3

wait 20

bystander packets
bystander mark
