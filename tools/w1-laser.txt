wait 40
/time set 6000
/gamerule doMobSpawning false
/gamerule doDaylightCycle false
/weather clear
/kill @e[type=slime]

bystander tick on

bystander fit lacrimal_laser

/summon pig 0.5 -60 7.5 {NoAI:1b}
bystander look 0 -4
/tp @s 5 -59 5.5 70 8

frames laser_by5_day 40
wait 3
bystander key secondary
wait 40

/time set 18000
