# Hopper, second pass: the jar's Hat Mount helmet in the collar, a stand-in getting out by the eject handle (the hoop
# opens and closes over the empty collar), the Jump Jets lifting a ridden Hopper, from a bystander's place
@observe 20
wait 60
/time set 6000
/gamerule doMobSpawning false
/gamerule doDaylightCycle false
/gamerule doWeatherCycle false
hud off
# --- a stand-in boards a Hopper; the real player watches it get out by the eject handle ---------------------------
mecha hopper 0 -60 3 hydraulic_legs,jump_jets,drill
bystander spawn 1 -60 1 0
headoffice trust add Bystander
bystander form jar
wait 5
bystander useentity nearest headinajar:hopper
wait 80
/tp @s 4.2 -58.3 7.2 140 14
wait 10
frames u_unboard 70
bystander key eject
wait 70
inspect near headinajar:hopper 6
bystander remove
# --- the player's own Hopper: the helmet on the jar, then the jets ----------------------------------------------
mecha hopper 0 -60 20 hydraulic_legs,jump_jets,drill
/tp @s 6.5 -60 17.5 0 0
jar
fit hat_mount
/item replace entity @s armor.head with minecraft:golden_helmet
wait 10
/tp @s 0.5 -60 18.9 0 -25
wait 5
key use
wait 70
view 2
look 0 10
wait 10
shot u_helmet_front
view 1
look 0 5
frames u_jets 90
key jump
wait 7
key jump
wait 80
inspect near headinajar:hopper 10
report second done
